function renderizarPlacar(jogadores) { const placar = document.getElementById("placar"); placar.innerHTML = ""; const titulo = document.createElement("h3"); titulo.textContent = "🏆 Placar"; placar.appendChild(titulo); jogadores .slice() .sort((a, b) => b.pontos - a.pontos) .forEach(jogador => { const linha = document.createElement("p"); linha.textContent = `${jogador.nome}: ${jogador.pontos} pontos`; placar.appendChild(linha); }); }
